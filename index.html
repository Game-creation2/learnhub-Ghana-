/* ============================== LEARNHUB GHANA - PART 3 Subjects, Quizzes and Questions ============================== */ /* Add a subject */ function addSubject() { if (!currentUser || currentUser.role !== "teacher") { alert("Only approved teachers can add subjects."); return; } const name = document.getElementById("subjectName").value.trim(); const topic = document.getElementById("subjectTopic").value.trim(); if (!name || !topic) { showMessage( "subjectMessage", "Please enter both subject and topic.", "error" ); return; } subjects.push({ id: Date.now().toString(), name: name, topic: topic, teacherId: currentUser.id, teacherName: currentUser.name, createdAt: new Date().toISOString() }); saveData(); document.getElementById("subjectName").value = ""; document.getElementById("subjectTopic").value = ""; showMessage( "subjectMessage", "Subject added successfully!", "success" ); loadLibrary(); } /* Add question box */ function addQuestionBox() { const builder = document.getElementById("questionBuilder"); const number = builder.children.length + 1; const box = document.createElement("div"); box.className = "question-box"; box.innerHTML = `

Question ${number}

`; builder.appendChild(box); } /* Create quiz */ function publishQuiz() { if (!currentUser || currentUser.role !== "teacher") { alert("Only approved teachers can publish quizzes."); return; } const title = document.getElementById("quizTitle").value.trim(); const subject = document.getElementById("quizSubject").value.trim(); const topic = document.getElementById("quizTopic").value.trim(); const classLevel = document.getElementById("quizClass").value.trim(); if (!title || !subject || !topic || !classLevel) { showMessage( "quizCreateMessage", "Please complete the quiz information.", "error" ); return; } const boxes = document.querySelectorAll("#questionBuilder .question-box"); if (boxes.length === 0) { showMessage( "quizCreateMessage", "Please add at least one question.", "error" ); return; } const questions = []; for (const box of boxes) { const question = box.querySelector(".questionText").value.trim(); const A = box.querySelector(".optionA").value.trim(); const B = box.querySelector(".optionB").value.trim(); const C = box.querySelector(".optionC").value.trim(); const D = box.querySelector(".optionD").value.trim(); const correct = box.querySelector(".correctAnswer").value; if (!question || !A || !B || !C || !D) { showMessage( "quizCreateMessage", "Please complete every question and all four options.", "error" ); return; } questions.push({ question: question, options: { A: A, B: B, C: C, D: D }, correct: correct }); } const quiz = { id: Date.now().toString(), title: title, subject: subject, topic: topic, classLevel: classLevel, teacherId: currentUser.id, teacherName: currentUser.name, questions: questions, createdAt: new Date().toISOString() }; quizzes.push(quiz); saveData(); document.getElementById("quizTitle").value = ""; document.getElementById("quizSubject").value = ""; document.getElementById("quizTopic").value = ""; document.getElementById("quizClass").value = ""; document.getElementById("questionBuilder").innerHTML = ""; showMessage( "quizCreateMessage", "🎉 Quiz published successfully!", "success" ); loadTeacherQuizzes(); loadAvailableQuizzes(); } /* Load teacher quizzes */ function loadTeacherQuizzes() { const box = document.getElementById("teacherQuizList"); if (!currentUser) { box.textContent = "Sign in to load."; return; } const mine = quizzes.filter( q => q.teacherId === currentUser.id ); if (mine.length === 0) { box.innerHTML = '

You have not created any quizzes yet.

'; return; } box.innerHTML = mine.map(q => `

${escapeHTML(q.title)}

${escapeHTML(q.subject)} ${escapeHTML(q.topic)} ${escapeHTML(q.classLevel)}

${q.questions.length} question(s)

`).join(""); } /* Load available quizzes */ function loadAvailableQuizzes() { const box = document.getElementById("availableQuizzes"); if (!currentUser) { box.textContent = "Sign in to view quizzes."; return; } if (quizzes.length === 0) { box.innerHTML = '

No quizzes have been published yet.

'; return; } box.innerHTML = quizzes.map(q => `

${escapeHTML(q.title)}

${escapeHTML(q.subject)} ${escapeHTML(q.topic)} ${escapeHTML(q.classLevel)}

Teacher: ${escapeHTML(q.teacherName)}

`).join(""); } /* Start quiz */ function startQuiz(quizId) { if (!currentUser) { alert("Please sign in first."); return; } const quiz = quizzes.find( q => q.id === quizId ); if (!quiz) { alert("Quiz not found."); return; } document.getElementById("quizPlayer") .classList.remove("hidden"); document.getElementById("playingQuizTitle") .textContent = quiz.title; document.getElementById("playingQuizInfo") .textContent = quiz.subject + " • " + quiz.topic + " • " + quiz.classLevel; const form = document.getElementById("dynamicQuizForm"); form.innerHTML = ""; quiz.questions.forEach((q, index) => { const box = document.createElement("div"); box.className = "question-box"; box.innerHTML = `

${index + 1}. ${escapeHTML(q.question)}

`; form.appendChild(box); }); form.dataset.quizId = quizId; document.getElementById("dynamicQuizResult") .textContent = ""; go("quizPlayer"); } /* Submit quiz */ function submitDynamicQuiz() { if (!currentUser) { alert("Please sign in first."); return; } const form = document.getElementById("dynamicQuizForm"); const quizId = form.dataset.quizId; const quiz = quizzes.find( q => q.id === quizId ); if (!quiz) { alert("Quiz not found."); return; } let score = 0; let answered = 0; quiz.questions.forEach((q, index) => { const selected = document.querySelector( `input[name="question_${index}"]:checked` ); if (selected) { answered++; if (selected.value === q.correct) { score++; } } }); if (answered < quiz.questions.length) { document.getElementById( "dynamicQuizResult" ).innerHTML = "Please answer all questions before submitting."; return; } const percentage = Math.round( (score / quiz.questions.length) * 100 ); results.push({ id: Date.now().toString(), quizId: quiz.id, quizTitle: quiz.title, studentId: currentUser.id, studentName: currentUser.name, teacherId: quiz.teacherId, score: score, total: quiz.questions.length, percentage: percentage, submittedAt: new Date().toISOString() }); saveData(); document.getElementById( "dynamicQuizResult" ).innerHTML = "🎉 You scored " + score + "/" + quiz.questions.length + " (" + percentage + "%)."; loadMyResults(); }/* ============================== LEARNHUB GHANA - PART 4 Library, Results, Classes and Teacher Approval ============================== */ /* Load learning library */ function loadLibrary() { const box = document.getElementById("libraryList"); if (!currentUser) { box.textContent = "Sign in to load materials."; return; } if (subjects.length === 0) { box.innerHTML = '

No learning materials have been added yet.

'; return; } box.innerHTML = subjects.map(subject => `

📚 ${escapeHTML(subject.name)}

Topic: ${escapeHTML(subject.topic)}

Added by ${escapeHTML(subject.teacherName)}

Learning Material
`).join(""); } /* Load student's results */ function loadMyResults() { const box = document.getElementById("myResults"); if (!currentUser || currentUser.role === "owner") { box.textContent = "Sign in to view results."; return; } const mine = results.filter( result => result.studentId === currentUser.id ); if (mine.length === 0) { box.innerHTML = '

You have not completed any quizzes yet.

'; return; } box.innerHTML = mine .slice() .reverse() .map(result => `

📝 ${escapeHTML(result.quizTitle)}

Score: ${result.score}/${result.total}

Percentage: ${result.percentage}%

Submitted: ${new Date(result.submittedAt).toLocaleString()}

`) .join(""); } /* Load teacher results */ function loadTeacherResults() { const box = document.getElementById("teacherResults"); if (!currentUser || currentUser.role !== "teacher") { box.textContent = "Sign in to load."; return; } const mine = results.filter( result => result.teacherId === currentUser.id ); if (mine.length === 0) { box.innerHTML = '

No students have completed your quizzes yet.

'; return; } box.innerHTML = mine .slice() .reverse() .map(result => `

👨‍🎓 ${escapeHTML(result.studentName)}

Quiz: ${escapeHTML(result.quizTitle)}

Score: ${result.score}/${result.total}

Percentage: ${result.percentage}%

${new Date(result.submittedAt).toLocaleString()}

`) .join(""); } /* ============================== LIVE CLASSES ============================== */ function loadClasses() { const box = document.getElementById("classList"); if (!currentUser) { box.textContent = "Sign in to load classes."; return; } const classes = JSON.parse( localStorage.getItem("learnhub_classes") || "[]" ); if (classes.length === 0) { box.innerHTML = `

No live classes have been scheduled yet.

`; return; } box.innerHTML = classes.map(liveClass => `

🎥 ${escapeHTML(liveClass.title)}

Teacher: ${escapeHTML(liveClass.teacherName)}

Date: ${escapeHTML(liveClass.date)}

Time: ${escapeHTML(liveClass.time)}

`).join(""); } /* Join live class */ function joinLiveClass(link) { if (!link) { alert("The teacher has not added a class link yet."); return; } window.open(link, "_blank"); } /* ============================== OWNER - TEACHER REQUESTS ============================== */ function loadTeacherRequests() { const box = document.getElementById("teacherRequests"); if (!currentUser || currentUser.role !== "owner") { box.textContent = "Owner access required."; return; } if (teacherRequests.length === 0) { box.innerHTML = '

There are no teacher requests.

'; return; } box.innerHTML = teacherRequests .slice() .reverse() .map(request => `

👤 ${escapeHTML(request.name)}

Email: ${escapeHTML(request.email)}

Status: ${escapeHTML(request.status)}

${ request.status === "pending" ? ` ` : "" }
`) .join(""); } /* Approve teacher */ function approveTeacher(requestId) { if (!currentUser || currentUser.role !== "owner") { alert("Owner access required."); return; } const request = teacherRequests.find( r => r.id === requestId ); if (!request) { alert("Teacher request not found."); return; } const user = users.find( u => u.id === request.userId ); if (!user) { alert("User account not found."); return; } user.role = "teacher"; user.teacherApproved = true; request.status = "approved"; saveData(); loadTeacherRequests(); alert( request.name + " has been approved as a teacher." ); } /* Reject teacher */ function rejectTeacher(requestId) { if (!currentUser || currentUser.role !== "owner") { alert("Owner access required."); return; } const request = teacherRequests.find( r => r.id === requestId ); if (!request) { alert("Teacher request not found."); return; } request.status = "rejected"; const user = users.find( u => u.id === request.userId ); if (user) { user.teacherApproved = false; } saveData(); loadTeacherRequests(); alert( request.name + " has been rejected." ); } /* ============================== STARTER DATA ============================== */ function createStarterData() { const classes = JSON.parse( localStorage.getItem("learnhub_classes") || "[]" ); if (!localStorage.getItem("learnhub_classes")) { localStorage.setItem( "learnhub_classes", JSON.stringify([ { id: "class1", title: "Introduction to Mathematics", teacherName: "LearnHub Teacher", date: "Saturday", time: "10:00 AM", link: "" } ]) ); } } /* ============================== PAGE STARTUP ============================== */ createStarterData(); window.addEventListener("load", function() { /* * Add the first question box * automatically for teachers. */ if ( document.getElementById("questionBuilder") && document.getElementById("questionBuilder").children.length === 0 ) { addQuestionBox(); } /* * Load any previously saved login. * This version keeps the user signed in * only during the current page session. */ loadLibrary(); loadAvailableQuizzes(); loadMyResults(); loadClasses(); }); /* ============================== MOBILE-FRIENDLY NAVIGATION ============================== */ document.querySelectorAll("nav a").forEach(link => { link.addEventListener("click", function(event) { const target = document.querySelector( this.getAttribute("href") ); if (target) { event.preventDefault(); target.scrollIntoView({ behavior: "smooth" }); } }); }); /* ============================== END OF PART 4 ============================== */
